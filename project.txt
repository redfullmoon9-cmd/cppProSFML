#include <iostream>
#include <SFML\Graphics.hpp>

sf::CircleShape drawCircle(float w, float h) {
    sf::CircleShape shape(30.f);
    shape.setFillColor(sf::Color::Green);
    shape.setPosition(0.0f, 0.0f);
    return shape; 
}


sf::RectangleShape drawRectangle(float w, float h) {
    sf::RectangleShape shape(sf::Vector2f{100.0f, 50.0f });
    //shape.setFillColor(sf::Color::Green);
    shape.setFillColor(sf::Color::Red);
    shape.setPosition(w/2.0f, h/2.0f);
    return shape; 
}

void setFontAndDraw(sf::RenderWindow window) {
    // 글을 표시하고 싶을 경우 폰트 로딩 필요
    sf::Font font;
    if (!font.loadFromFile("../resources/font/arial.ttf")){
        std::cout << "Font Loading Error!" << std::endl;
    }

    // 화면에 그릴 텍스트 객체 생성
    sf::Text text;
    text.setFont(font);
    text.setString("Hello SFML!");
    text.setCharacterSize(24);
    //text.setFillColor(sf::Color::White);
    text.setFillColor(sf::Color::Blue);

    window.draw(text); 
}

int main()
{
    float screen_width = 1024; 
    float screen_height = 768; 
    float direction = 1.0f; 
    // 화면을 그릴 윈도우 생성
    sf::RenderWindow window(sf::VideoMode(screen_width, screen_height), "SFML works!");

    // 화면에 그릴 원 객체 생성
    sf::CircleShape shape= drawCircle(screen_width, screen_height); 
    sf::RectangleShape rec = drawRectangle(screen_width, screen_height); 

    float speed = 10; 

    // 윈도우가 열려 있을 동안 반복 
    while (window.isOpen())
    {
        // 종료 조건
        sf::Event event;
        while (window.pollEvent(event))
        {
            if (event.type == sf::Event::Closed)
                window.close();
        }

        //--- 화면을 그리는 세 단계
        // 1) 이미 그려져 있는 모든 것을 지운다.
        window.clear();
        
        if (speed < 0) {
            speed = 0; 
            direction = +1.0f; 
        }
        else if (speed > screen_width) {
            speed = 650; 
            direction= -1.0f; 
        }

        speed += 0.1f*direction; 
        shape.setPosition(speed, 0.0f); 
        rec.setPosition(0.0f, speed); 

        // 2) 그리고 싶은 것들을 draw 함수를 사용해 그린다.
        window.draw(shape);
        window.draw(rec);

        // 3) 화면에 표시한다.
        window.display();
    }

    return 0;

}

26. 2.13 CppProSFML - 방향이동시 좀더 부드럽게

<수학적 이론 -atan2(y, x)의 원리 > 
게임에서의 활용: 캐릭터가 바라볼 방향 벡터 (x, y)가 있을 때, atan2(y, x)를 사용하면 
해당 방향의 절대 각도를 얻을 수 있습니다.

SFML 주의사항: SFML의 좌표계는 Y축이 아래로 갈수록 커지므로, 
일반적인 수학 좌표계와 반대로 생각해야 합니다. 
또한 SFML의 0도는 오른쪽(Positive X)을 향합니다.

atan(a): 1개의 인자(tan 값)를 사용하여  범위 내의 값만 반환.
atan2(y, x): 2개의 인자를 사용하여, x가 -π/2 ~ π/2 경우를 포함한 전체 범위를 올바르게 계산. 

**현재 보정값을 넣지 않으면 각도가 제대로 맞지 않는 문제.
